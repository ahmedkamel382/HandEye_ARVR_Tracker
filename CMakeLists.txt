cmake_minimum_required(VERSION 3.15)
project(GradProject CXX)

# Force modern pure C++17 standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Point CMake directly to the vcpkg built package files
set(OpenCV_ROOT "${CMAKE_CURRENT_BINARY_DIR}/vcpkg_installed/x64-windows/share/opencv4")

# 1. Locate OpenCV (Provided by vcpkg)
find_package(OpenCV CONFIG REQUIRED)

# 2. Locate the local Python Environment 
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)

# 3. Ask Python exactly where Pybind11 is installed
execute_process(
    COMMAND "${Python3_EXECUTABLE}" -c "import pybind11; print(pybind11.get_cmake_dir())"
    OUTPUT_VARIABLE pybind11_DIR
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

# 4. Locate Pybind11 using the directory we just found
find_package(pybind11 REQUIRED)

# 4b. Lock-free queue (moodycamel concurrentqueue v1.0.4). Header-only and VENDORED in
#     core_engine/third_party, so the build needs no network or git.
add_library(core_concurrentqueue INTERFACE)
target_include_directories(core_concurrentqueue SYSTEM INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/core_engine/third_party/concurrentqueue")

# 4c. Tell the C++ exe where Python lives, so the embedded interpreter can find
#     the venv's packages (mediapipe, numpy, scikit-learn, ...). Paths are baked
#     into a generated header; override at runtime with --ai-root / --site.
execute_process(
    COMMAND "${Python3_EXECUTABLE}" -c "import sys; print(sys.base_prefix)"
    OUTPUT_VARIABLE PY_HOME_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(
    COMMAND "${Python3_EXECUTABLE}" -c "import sysconfig; print(sysconfig.get_paths()['purelib'])"
    OUTPUT_VARIABLE PY_SITE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
file(TO_CMAKE_PATH "${PY_HOME_DIR}" PY_HOME_DIR)   # forward slashes: safe inside C string literals
file(TO_CMAKE_PATH "${PY_SITE_DIR}" PY_SITE_DIR)
set(AI_PIPELINE_DIR "${CMAKE_SOURCE_DIR}/ai_pipeline")
configure_file(core_engine/include/build_paths.hpp.in
               "${CMAKE_BINARY_DIR}/generated/build_paths.hpp" @ONLY)

# 5. Define the Core Engine Executable
add_executable(core_engine core_engine/src/main.cpp)
target_include_directories(core_engine PRIVATE
    core_engine/include
    "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(core_engine PRIVATE NOMINMAX)

# 6. Link OpenCV, the Pybind11 Embed module and the lock-free queue
target_link_libraries(core_engine PRIVATE ${OpenCV_LIBS} pybind11::embed core_concurrentqueue)

# 6b. The two Random Forests (gaze blink/wink classifier, hand gesture classifier) compiled to native C by
#     Treelite/TL2cgen (strongly recommended). Produced by:
#         python core_engine/tools/export_forests.py        (re-run whenever a .pkl is retrained)
#     Without them the engine falls back to scikit-learn through Python: several ms per call instead of microseconds.
#     Every global symbol of the generated code is prefixed (gaze_ / hand_) by the export script, so both forests
#     link into one executable; forest_shim.c is compiled once per forest with -DFOREST_PREFIX=...
# (enable_language must run at top level, not inside the function below)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/core_engine/generated/gaze_forest/main.c" OR
   EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/core_engine/generated/hand_forest/main.c")
    enable_language(C)
endif()
function(core_add_forest NAME DIR DEFINE)
    set(${DEFINE} OFF PARENT_SCOPE)
    if(EXISTS "${DIR}/main.c" AND EXISTS "${DIR}/header.h")
        file(STRINGS "${DIR}/header.h" _hdr REGEX "${NAME}_predict")
        if(NOT _hdr)
            message(FATAL_ERROR "${DIR} was generated by an older export script (symbols are not prefixed). "
                                "Run  python core_engine/tools/export_forests.py  and re-configure.")
        endif()
        file(GLOB _srcs CONFIGURE_DEPENDS "${DIR}/*.c")
        add_library(${NAME}_forest STATIC ${_srcs} core_engine/src/forest_shim.c)
        target_include_directories(${NAME}_forest PRIVATE "${DIR}")
        target_compile_definitions(${NAME}_forest PRIVATE FOREST_PREFIX=${NAME}_)
        target_link_libraries(core_engine PRIVATE ${NAME}_forest)
        target_compile_definitions(core_engine PRIVATE ${DEFINE}=1)
        set(${DEFINE} ON PARENT_SCOPE)
        message(STATUS "${NAME} forest: Treelite-compiled native C (${DIR})")
    else()
        message(WARNING "${NAME} forest: NOT compiled. The engine will use scikit-learn through Python "
                        "(gaze: slow; hand: slow, or a rough geometric fallback if hand_gesture_rf.pkl is missing). "
                        "Run  python core_engine/tools/export_forests.py  and re-configure.")
    endif()
endfunction()
core_add_forest(gaze "${CMAKE_CURRENT_SOURCE_DIR}/core_engine/generated/gaze_forest" CORE_HAVE_TREELITE_FOREST)
core_add_forest(hand "${CMAKE_CURRENT_SOURCE_DIR}/core_engine/generated/hand_forest" CORE_HAVE_HAND_FOREST)

# 6c. Configure-time pre-flight checks for Python packages, MediaPipe models, and your Gaze/Wink assets.
foreach(_pkg numpy mediapipe sklearn cv2)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import ${_pkg}"
                    RESULT_VARIABLE _rc OUTPUT_QUIET ERROR_QUIET)
    if(NOT _rc EQUAL 0)
        message(WARNING "Python package '${_pkg}' cannot be imported by ${Python3_EXECUTABLE}. "
                        "Install the AI requirements into that venv (pip install numpy mediapipe scikit-learn opencv-python).")
    endif()
endforeach()

foreach(_m face_landmarker.task hand_landmarker.task)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/assets/${_m}")
        message(WARNING "Model file assets/${_m} is missing - the engine will fail at startup until it is added.")
    endif()
endforeach()

if(NOT EXISTS "${CMAKE_SOURCE_DIR}/assets/hand_gesture_rf.pkl")
    message(WARNING "assets/hand_gesture_rf.pkl is missing - hand gestures would use a rough geometric fallback "
                    "(train the hand model with Engineer 2's model_trainer.py).")
endif()

# Explicit checks for your gaze and wink components / exported forest directories
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/core_engine/generated/gaze_forest")
    message(WARNING "Gaze/wink native forest folder is missing under core_engine/generated/gaze_forest. "
                    "Run 'python core_engine/tools/export_forests.py' to generate native C inference files for your gaze/wink models.")
endif()

if(MSVC)
    target_compile_options(core_engine PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(core_engine PRIVATE -Wall -Wextra -fvisibility=hidden)
endif()

# 7. Windows: the embedded interpreter needs python3xx.dll next to the exe
if(WIN32)
    file(GLOB PY_RUNTIME_DLLS "${PY_HOME_DIR}/python3*.dll")
    foreach(dll ${PY_RUNTIME_DLLS})
        add_custom_command(TARGET core_engine POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:core_engine>")
    endforeach()
endif()

include(CTest)
add_subdirectory(dsp_filter)                # Engineer 4: defines the static library target "dsp_filter"
add_subdirectory(core_engine/os_integration)   # Engineer 5: OS integration + tray app (targets os_integration, os_integration_tray, ...)

# 7b. Eng 5's OS integration is the LAST stage of the pipeline (after the DSP filter): screen mapping, gestures, clicks,
#     SendInput, tray icon and calibration wizard. The engine feeds it through core_engine/include/os_bridge.hpp.
#     Windows only; elsewhere the bridge is a no-op stub and the engine only tracks.
if(WIN32 AND TARGET os_integration_tray)
    target_link_libraries(core_engine PRIVATE os_integration_tray)
    target_compile_definitions(core_engine PRIVATE CORE_HAVE_OS_INTEGRATION=1)
    message(STATUS "OS stage: os_integration (Engineer 5)")
else()
    message(WARNING "os_integration_tray is not available (Windows only): the engine will track but not move the mouse.")
endif()

# 8. Use Eng 4's 1-Euro filter as the smoothing stage (the engine falls back to its built-in filter without it).
if(TARGET dsp_filter)
    target_link_libraries(core_engine PRIVATE dsp_filter)
    target_compile_definitions(core_engine PRIVATE CORE_HAVE_DSP_FILTER=1)
    message(STATUS "Smoothing filter: dsp_filter (Engineer 4)")
endif()
